public static float[] getAutocorrelation(float[] projection) {
    float[] rv = new float[projection.length - 1];
    for (int i = 1; i < projection.length - 1; i++) {
        rv[i] = (projection[i] * projection[i - 1]) / 100f;
    }
    return rv;
}
